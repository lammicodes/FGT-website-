const express = require('express'); const cron = require('node-cron'); const app = express(); app.use(express.json()); // ========================================== // 1. SERVICES DATA STRUCTURE (TAJAAJILOOTA GUUTUU) // ========================================== const servicesList = [ { id: 1, title: "Tutoring & any educational writing Services (Online & Physical)", category: "education", status: "active" }, { id: 2, title: "Advertising services for any business organization", category: "advertising", status: "active" }, { id: 3, title: "Existing Service 3", // Service keessan isa 3ffaa kan duraan ture category: "general", status: "active" }, { id: 4, title: "Digital E-commerce & Market Place", category: "ecommerce", status: "active", rules: { maxMediaAllowed: 4, pricingPlans: [ { durationMonths: 3, priceETB: 100 }, { durationMonths: 6, priceETB: 150 }, { durationMonths: 12, priceETB: 200 } ] } } ]; // Mock Database (Post-oota Store keessatti kuufaman) let storePostsDatabase = []; // ========================================== // 2. API ENDPOINTS (ROUTERS) // ========================================== // A. Tarree Tajaajiloota Hunda Agarsiisuu (Get All Services) app.get('/api/services', (req, res) => { return res.status(200).json({ success: true, data: servicesList }); }); // B. Tajaajila 4ffaa: Store Post Haaraa Creat Gochuu app.post('/api/ecommerce/posts', (req, res) => { const { title, description, mediaList, subscriptionMonths } = req.body; // OFEEGGANNOO 1: Suuraa/Viidiyo 4 ol akka hin darbite mirkaneessuu if (!mediaList || !Array.isArray(mediaList) || mediaList.length === 0) { return res.status(400).json({ success: false, message: "Dogoggora: Suuraa ykn viidiyo meeshaa keessanii kamoo dhiyeessaa." }); } if (mediaList.length > 4) { return res.status(400).json({ success: false, message: "Ofeeggannoo: Suuraa ykn viidiyo 4 olii fe'uun dhowwaadha!" }); } // OFEEGGANNOO 2: Ibsa (Description) guutuu galchuu if (!description || description.trim().length < 10) { return res.status(400).json({ success: false, message: "Ofeeggannoo: Ibsa (description) guutuu meeshaa keessanii galchuun dorgomaadha!" }); } // OFEEGGANNOO 3: Karoora Kaffaltii Mirkaneessuu const plans = { 3: 100, 6: 150, 12: 200 }; const pricePaid = plans[subscriptionMonths]; if (!pricePaid) { return res.status(400).json({ success: false, message: "Karoora Kaffaltii Sirrii Filadhaa: Baatii 3 (100 Birr), Baatii 6 (150 Birr), ykn Waggaa 1 (200 Birr)." }); } // Guyyaa Dhumaa (Expiry Date) Shallaguu const createdAt = new Date(); const expiresAt = new Date(); expiresAt.setMonth(createdAt.getMonth() + Number(subscriptionMonths)); const newPost = { id: `POST-${Date.now()}`, title, description, mediaList, pricePaid, subscriptionMonths, createdAt: createdAt.toISOString(), expiresAt: expiresAt.toISOString(), isActive: true }; // Database-saniitti dabaluu storePostsDatabase.push(newPost); return res.status(201).json({ success: true, message: "Maxxansi keessan milkaa'inaan raawwatameera!", data: newPost }); }); // C. Maxxansoota Hojii Irra Jiran (Active Posts) Qofa Agarsiisuu app.get('/api/ecommerce/posts', (req, res) => { const activePosts = storePostsDatabase.filter(post => post.isActive); return res.status(200).json({ success: true, data: activePosts }); }); // ========================================== // 3. AUTOMATIC REMOVAL TASK (CRON JOB) // ========================================== // Halkan sa'a 6:00 (00:00) irratti post-oota guyyaan waliigaltee dhume ofumaan DELETE/INACTIVE godha cron.schedule('0 0 * * *', () => { console.log("⏰ Automatic Expiration Check Run..."); const now = new Date(); storePostsDatabase.forEach(post => { if (new Date(post.expiresAt) <= now && post.isActive) { post.isActive = false; console.log(`❌ Post ID: ${post.id} guyyaan waliigaltee dhumatee sirna irraa haqameera.`); } }); }); // Server Start const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server-niin keessan port ${PORT} irratti hojjechaa jira.`); });
